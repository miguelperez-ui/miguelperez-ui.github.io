📡 XMLHttpRequest (nativo)
GET — Recibir datos
function solicitud() {
  const xhr = new XMLHttpRequest();
  xhr.open("GET", "https://api.com/users");
  xhr.responseType = "json"; // auto-parsea
  xhr.onload = function() {
    if (xhr.readyState == 4 && xhr.status == 200) {
      console.log(xhr.response); // objeto JS
    }
  };
  xhr.send();
}
POST — Enviar datos
function envio(obj) {
  const xhr = new XMLHttpRequest();
  xhr.open("POST", "https://api.com/posts");
  xhr.responseType = "json";
  xhr.onload = function() {
    if (xhr.readyState == 4 && xhr.status == 201) {
      console.log(xhr.response);
    }
  };
  xhr.send(JSON.stringify(obj));
}

📨 AJAX con jQuery ($.ajax)
GET — Recibir
$.ajax({
  url:      "https://api.com/users",
  method:   "GET",
  dataType: "json"  // auto-parsea respuesta
}).done(response => console.log(response))
  .fail(err => console.log(err));
POST — Enviar
$.ajax({
  url:         "https://api.com/posts",
  method:      "POST",
  data:        JSON.stringify(miObjeto),
  contentType: "application/json", // lo que envío
  dataType:    "json"             // lo que espero
}).done(response => console.log(response))
  .fail(err => console.log(err));
DELETE
$.ajax({
  url:    "https://api.com/items?id=" + id,
  method: "DELETE",

📋 Patrón: Array + generaTabla + splice
// 1. Array global (base de datos local)
let lista = [];
let contadorId = 0;

// 2. Insertar en array y regenerar tabla
function insertar(obj) {
  lista.push(obj);          // Añade al final del array
  generaTabla(lista);       // Reconstruye toda la tabla
}

// 3. Reconstruir tabla desde el array
function generaTabla(lista) {
  $("#tbody").html("");    // Vaciamos primero
  lista.forEach(function(obj, indice) {
    $("#tbody").append(generaTr(obj, indice));
  });
}

// 4. Generar TR con índice guardado en data-indice
function generaTr(obj, indice) {
  return `<tr id="fila-${obj.id}" data-indice="${indice}"
         onclick="eliminarFila(this)">
    <td>${obj.nombre}</td>
    <td>${obj.apellidos}</td>
    <td>${obj.edad}</td>
    <td>${obj.ciudad}</td>
  </tr>`;
}

// 5. Eliminar del array (splice) y regenerar
function eliminarFila(tr) {
  const indice = parseInt($(tr).attr("data-indice"));
  lista.splice(indice, 1); // splice(pos, cantidad)
  generaTabla(lista);      // Índices recalculados ✅
}
✅ Regenerar siempre tras splice porque los índices del array cambian.
